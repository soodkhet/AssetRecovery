### 0.4 วิธีแยก "สิทธิ์" ออกจาก "business" (งวด ต.ค. locked) 📄
- ลำดับในทุก route (📄 `lib/auth/require-permission.ts`, `withApiPermission`/`withEndpoint` ใน `lib/api/http.ts`, `withPermission`): **session (401) → สถานะบัญชี/อายุ session/ต้องเปลี่ยนรหัส (401/403) → capability (403 `PERMISSION_DENIED`) → (handler) zod 400 → โหลดแถว + scope (404) → business/ยามงวด (400 `PERIOD_LOCKED_DIRECT_EDIT` ฯลฯ)**
- จัดผลเป็น 6 ชนิด (ฟังก์ชัน `classify()` §0.7):

| ชนิด | เงื่อนไข | ความหมาย |
|---|---|---|
| `D` | 403 + `PERMISSION_DENIED` | ถูกปัดที่ชั้นสิทธิ์ (capability หรือ scope ทั้งก้อน) |
| `A` | 2xx / 400 / 404-ที่มี `error.code` / 409 / 422 | **ผ่านชั้นสิทธิ์แล้ว** — 400 `PERIOD_LOCKED_DIRECT_EDIT` ก็นับ `A` (แปลว่าผ่าน capability มาถึงยามงวดแล้ว) |
| `B` | 403 code อื่น (`UNLOCK_REQUIRES_EXECUTIVE`, `INSUFFICIENT_APPROVAL_LEVEL`, `SEGREGATION_OF_DUTIES_VIOLATION`) | ผ่าน route แต่ business ปัดเรื่องระดับอนุมัติ — นับว่าตรงเมื่อคาด `A` |
| `N` | 404/405 ที่ไม่มี `error.code` | ไม่มี route (เช่น `/api/portal/*`) |
| `AUTH` | 401 | session หลุด/หมดอายุ (24 ชม. จาก `last_login_at`) → login ใหม่แล้วรันซ้ำ ไม่ใช่บั๊ก |
| `ERR` | 5xx | 🐞 S2 ทุกตัว |
- **เพราะ write probe ส่ง `{}` หรือ id สุ่ม** ผู้ที่มีสิทธิ์จะได้ 400/404 ก่อนถึงยามงวด ⇒ ไม่มีทางเกิด mutation · ยามงวด locked เป็น "ตาข่ายชั้นสอง" ใน R10.08 (claim วันที่ 05/10/2569) — ได้ 403 = ปัดที่สิทธิ์ · ได้ 400 `PERIOD_LOCKED_DIRECT_EDIT` = ผ่านสิทธิ์ ถูกปัดที่งวด
- คาด `D` แต่ได้ 400 (validation) = **🐞 ลำดับผิด: ตรวจ body ก่อนสิทธิ์** (เคยมีจริงที่ `POST /api/reports/:id/export` — R10-N3)

### 0.5 Error code ที่คาด (`docs/24` · `lib/auth/errors.ts` · `lib/roles/errors.ts`)
| code | HTTP | เกิดเมื่อ |
|---|---|---|
| `UNAUTHENTICATED` / `SESSION_EXPIRED` | 401 | ไม่มี cookie / login เกิน 24 ชม. / หลัง logout |
| `INVALID_CREDENTIALS` | 401 | รหัสผิด **และ** ไม่มีผู้ใช้ — ต้องได้ body เดียวกัน (📄 `auth-service.ts` "ตอบเหมือนรหัสผิดทุกประการ") |

| `CASE_NOT_FOUND` · `ASSET_NOT_FOUND` · `BILLING_BATCH_NOT_FOUND` · `ASSIGNMENT_NOT_FOUND` · `PAYEE_NOT_FOUND` · `USER_NOT_FOUND` ฯลฯ | 404 | ไม่มีจริง **หรืออยู่นอก scope** (ต้องแยกไม่ออก) |
| `CAPABILITY_LOCKED` | 400 | มอบ/แก้ capability ที่ติด 🔒 (9 รายการ) |
| `ROLE_NOT_EDITABLE` | 400 | แก้สิทธิ์ role ที่ `is_editable=false` (14 จาก 15 role — มีแค่ **ธุรการ** ที่แก้ได้) |
| `CAPABILITY_NOT_FOUND` | 400 | รหัส capability ไม่มีในระบบ (ใช้เป็นตัวกันพลาดใน R10.22) |
| `SEED_ROLE_DELETE` / `ROLE_IN_USE` | 400 | ลบ seed role / role ที่ยังมีผู้ใช้ |
| `REPORT_NOT_FOUND` | 404 | id รายงานผิด |
| `UNLOCK_REQUIRES_EXECUTIVE` | 403 | บัญชีผ่านชั้น route ของ unlock แต่ไม่ใช่ผู้บริหาร (R8 พิสูจน์แล้ว) |

**`_h.mjs`**
```js
// ตัวช่วย R10 v3 — API ขนาน (ไม่เปิด browser) · ห้าม log รหัสผ่าน/ body ของ login
import { request } from '@playwright/test'
import { execFileSync } from 'node:child_process'
import { appendFileSync, mkdirSync } from 'node:fs'
import { randomUUID } from 'node:crypto'
import { BASE, credentials } from '../lib.mjs'
export { BASE }
mkdirSync('uat/bin/r10v3', { recursive: true })
const LOG = 'uat/bin/r10v3/run.log'
export const rnd = () => randomUUID()
export function log(...a) { const s = a.map(x => (typeof x === 'string' ? x : JSON.stringify(x))).join(' '); console.log(s); appendFileSync(LOG, s + '\n') }
export function q(sql) { return execFileSync('uat/bin/q.sh', [sql], { encoding: 'utf8' }).replace(/^SET\n/, '').trim() }
/** login ผ่าน API (ได้ cookie jar ของตัวเอง) + เก็บ storageState ให้ openAs() ใช้ต่อ — body ต้องเป็น {identifier,password} */
export async function login(u, { save = true } = {}) {
  const c = credentials(u)
  const ctx = await request.newContext({ baseURL: BASE })
  const r = await ctx.post('/api/auth/login', { data: { identifier: c.username, password: c.password }, failOnStatusCode: false })
  if (r.status() !== 200) throw new Error(`login ${u} → ${r.status()}`)
  if (save) await ctx.storageState({ path: `uat/.auth/${u}.json` })
  return ctx
}
export async function call(ctx, method, path, data) {
  const t = Date.now()
  const r = await ctx.fetch(path, { method, ...(data === undefined ? {} : { data }), failOnStatusCode: false, maxRedirects: 0 })
  let body = null
  try { body = await r.json() } catch {}
  return { status: r.status(), code: body?.error?.code ?? null, msg: body?.error?.message ?? null, body, ms: Date.now() - t }
}
export function classify(res) {
  if (res.status === 401) return 'AUTH'
  if (res.status >= 500) return 'ERR'
  if (res.status === 403) return res.code === 'PERMISSION_DENIED' ? 'D' : 'B'
  if ((res.status === 404 || res.status === 405) && res.code === null) return 'N'
  return 'A'
}
/** ปฏิเสธแบบไม่ leak: status + code + message + ชุด key ของ error ต้องเหมือน id สุ่มทุกประการ */
export function sameDenial(a, b) {
  const keys = (x) => Object.keys(x.body?.error ?? {}).sort().join(',')
  return a.status === b.status && a.code === b.code && a.msg === b.msg && keys(a) === keys(b)
}
export async function pool(items, n, fn) { const qu = [...items]; await Promise.all(Array.from({ length: n }, async () => { while (qu.length) await fn(qu.shift()) })) }

```
